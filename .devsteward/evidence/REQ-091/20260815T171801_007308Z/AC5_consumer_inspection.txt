# DevSteward per-project configuration (lives in the consumer repo).
# The engine is shared and pinned by tag; this file is yours to edit.

profile: req

# Where the engine's documents live, relative to the repo root. All four are yours to
# place: a project whose `docs/` is owned by a docs generator (mkdocs, Sphinx, Docusaurus
# — everything under it gets built and deployed) keeps the engine's artifacts out of that
# tree by pointing these elsewhere, e.g. `requirements/`, `requirements/plans/`.
#   requirements_dir — the REQ corpus (`steward lint`/`status` read it)
#   index_file       — REQUIREMENTS_INDEX.md; status source of truth with the frontmatter
#   plans_dir        — plan artifacts; the land refuses a REQ no file here names (REQ-029)
#   concepts_dir     — concept deliverables for a REQ with `process.concept` (REQ-039)
requirements_dir: docs/requirements
index_file: docs/requirements/REQUIREMENTS_INDEX.md
plans_dir: docs/plans
concepts_dir: docs/concepts

# Account / quota provider for headless runs.
#   clauder — delegate the per-step budget gate to the external `clauder` CLI (REQ-058):
#             `clauder gate` decides proceed/switch/wait against the combined budget across
#             accounts and tells cswap when to switch. The engine never touches cswap
#             directly and never runs `clauder monitor` (operator-started, out-of-band).
#             Degrades to "proceed, no quota check" when `clauder` is absent from PATH.
#   single  — plain `claude`, no quota machinery.
accounts:
  provider: clauder

# How a headless `claude -p` run is allowed to act on the repo. Headless sessions have
# no interactive approver, so without this every edit stalls and the run does nothing.
#   dangerously-skip  — bypass all permission checks (autonomous edits; the default,
#                       matches the proven batch runner). Use in a trusted repo.
#   bypassPermissions / acceptEdits / plan / default — claude --permission-mode values.
#   ask               — no flag (attended/interactive; a headless run can't edit).
#
# Per-step-kind model/effort (REQ-029/030). Each REQ is one `develop` Claude session; a
# red gate spawns up to two `repair` sessions, then parks; a REQ with artifact/manual
# acceptance criteria adds one `validate` System Tester session. `claude.model`/
# `claude.effort` are the flat defaults (Opus / high); `claude.steps.<kind>` overrides a
# kind. Built-in defaults: develop and validate inherit the flat values; repair drops to
# Sonnet (cheap cold restart).
#   steps:
#     develop:  {model: claude-opus-4-8, effort: high}
#     repair:   {model: claude-sonnet-4-6}
#     validate: {model: claude-opus-4-8, effort: high}
claude:
  permission_mode: dangerously-skip
  model: claude-opus-5
  effort: high
  steps:
    repair:   {model: claude-sonnet-5}
    validate: {model: claude-sonnet-5}

# Branching model — trunk-based. `main` is production (release tags are cut here); the
# integration branch is the default working branch. All work — declaration *and*
# implementation — lands on the integration branch; there are no feature branches, no
# worktrees, and the engine never switches branches. The engine refuses to autocommit on
# `production_branch` — switch to the integration branch to run. The only branch operation
# is the human-gated integration → production release PR.
git:
  production_branch: main
  integration_branch: dev

# How the land gate verifies (REQ-028). A `land` step is DONE only when its named
# acceptance behaviour *observably ran and passed*: a skipped or zero-collection named
# test fails the gate (skip ≠ green), and the whole suite must be clean.
#   full_suite — the command for the project's full test suite, run at land in addition
#                to each named acceptance test (default `python -m pytest`; set to `null`
#                to disable). Skips in the full suite stay legal; failures/errors fail land.
#   python     — optional explicit interpreter the gate runs tests under (e.g.
#                `.venv/bin/python`). A configured-but-unusable interpreter is a hard,
#                surfaced error. Unset: discover the project venv, then the engine's own.
#   env_file   — the declared env-file the land's commit-integrity self-check carries into
#                its ephemeral tree extract (REQ-072), so the check runs in the same
#                declared environment the gate used (default `.env`; set to `null` to
#                disable). Honor-when-present: an absent file is a no-op. The file is
#                copied only into the throwaway extract dir — its contents never reach a
#                message, event, or artifact.
verify:
  full_suite: bash -lc 'cargo test && python -m pytest'
  # No project virtualenv: pytest is installed system-wide, and the engine's own pipx venv
  # cannot import it. Point the gate at the system interpreter explicitly.
  python: /usr/bin/python3
#   env_file: .env

#   ask               — no flag (attended/interactive; a headless run can't edit).

  permission_mode: dangerously-skip

  # ── Model & effort: which Claude runs a headless step, and how hard it thinks ──
  #
  # Two layers:
  #   1. A FLAT default directly under `claude:` — applies to every headless step
  #      unless a per-step override says otherwise. Unset = the claude CLI's own
  #      default (whatever `claude` resolves to on the box).
  #   2. Per-step-kind overrides under `claude.steps.<kind>:` — <kind> is the step
  #      the engine is driving: `develop` (the design+build session) or `validate`
  #      (the fresh-context System-Test session). An override wins over the flat
  #      default for that kind only.
  #
  # Keys (same at either layer):
  #   model:  passed through to `claude --model`. Aliases the CLI resolves
  #           (`fable`, `opus`, `sonnet`, `haiku`) or full ids
  #           (`claude-fable-5`, `claude-opus-4-8`, ...).
  #   effort: reasoning effort — `low` | `medium` | `high`. More effort = more
  #           thinking tokens = better on cross-cutting/invariant-heavy work,
  #           more budget burned on routine work.
  #
  # House heuristic (2026-07-12): pattern-following REQs with golden-pinned ACs
  # (e.g. REQ-106/109, the planner lineage) → opus + high is enough; architecturally
  # dense REQs touching many subsystems/invariants (e.g. REQ-107) → fable + medium
  # earns its cost. Set the override BEFORE `steward run`/`advance`, revert after —
  # this file is committed, so a per-REQ choice left behind silently becomes the
  # default for the next REQ.
  #
  # Flat default — pinned (REQ-091). It must stay set: a kind with no override below
  # resolves to this, and since devsteward REQ-091 an unattended spawn with no model at
  # all refuses rather than silently taking claude's own default.

  steps:

    # an invariant-heavy REQ:

       effort: medium

    # validate — the System-Test session starts fresh-context, so high effort buys
    # little there; medium cuts budget.
    validate:
      model: sonnet
      effort: medium
